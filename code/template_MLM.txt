[[init]]
iterators=sample;
sample=1:200;

filename="decay_[[sample]].inp";
outputDirectory=multilevel;
[[/init]]
TITLE: multilevel_decay_[[sample]]
DATA: 
   FILE = data[[sample]].dat;
VARIABLE:
   NAMES = id x1-x3 m1-m3 y1-y3 time t;
   USEVAR = id x1-x3 m1-m3 y1-y3;
   cluster = id;
   
ANALYSIS:
  TYPE=TWOLEVEL;
  ESTIMATOR = BAYES;
  PROCESSORS = 3;
  FBITERATIONS = 10000;
  ALGORITHM=GIBBS(RW);
  POINT=MEAN;

MODEL:
%between%
    tx by x1-x3 (lx1-lx3);
	tm by m1-m3 (lm1-lm3);
	ty by y1-y3 (ly1-ly3);
    [x1-x3@0];
    [m1-m3@0];
    [y1-y3@0];
    x1-x3@0;
    m1-m3@0;
    y1-y3@0;
    tx@0;
	tm on tx (a);
    [tm] (mu);
	ty on tx (c)
		  tm (b);
    [ty] (nu);
%within%
    wx by x1-x3 (lx1-lx3);
	wm by m1-m3 (lm1-lm3);
	wy by y1-y3 (ly1-ly3);
    wm on wx (a);
    [wm] (mu);
    wy on wx (c)
          wm (b);
    [wy] (nu);
MODEL CONSTRAINT:
	new (ind);
	ind=a*b;
